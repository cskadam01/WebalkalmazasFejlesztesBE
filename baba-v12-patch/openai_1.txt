    private static EmploymentExtraction parseEmploymentExtraction(String raw) throws Exception {
        String t = raw == null ? "" : raw.trim();
        if (t.startsWith("```")) {
            int firstBreak=t.indexOf('\n'); if(firstBreak>=0)t=t.substring(firstBreak+1);
            int endFence=t.lastIndexOf("```"); if(endFence>=0)t=t.substring(0,endFence);
            t=t.trim();
        }
        int a=t.indexOf('{'), b=t.lastIndexOf('}');
        if(a>=0 && b>a) t=t.substring(a,b+1);
        JSONObject j=new JSONObject(t);
        EmploymentExtraction e=new EmploymentExtraction();
        e.employer=j.optString("employer","").trim();
        e.start=j.optString("start","").trim();
        e.end=j.optString("end","").trim();
        e.ongoing=j.optBoolean("ongoing",false);
        e.monthlyGross=j.optLong("monthlyGross",0);
        e.sickPayDays=j.has("sickPayDays")?j.optInt("sickPayDays",-1):-1;
        e.unpaidDays=j.has("unpaidDays")?j.optInt("unpaidDays",-1):-1;
        e.insured=j.has("insured")?j.optBoolean("insured",true):true;
        e.notes=j.optString("notes","").trim();
        return e;
    }
