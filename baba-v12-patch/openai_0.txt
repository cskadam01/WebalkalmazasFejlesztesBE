    public static EmploymentExtraction extractEmploymentData(Context context, Uri uri, String apiKey, String model) throws Exception {
        String fileName = getFileName(context.getContentResolver(), uri);
        String mime = context.getContentResolver().getType(uri);
        if (mime == null) mime = "application/octet-stream";
        String fileId = uploadFile(context, uri, fileName, mime, apiKey);
        String text = createStructuredResponse(fileId, apiKey, model);
        return parseEmploymentExtraction(text);
    }

        String fileId = uploadFile(context, uri, fileName, mime, apiKey);

    private static String uploadFile(Context context, Uri uri, String fileName, String mime, String apiKey) throws Exception {
        String boundary = "----BabaEllatas" + System.currentTimeMillis();
        HttpURLConnection c = (HttpURLConnection) new URL("https://api.openai.com/v1/files").openConnection();
        c.setRequestMethod("POST"); c.setDoOutput(true); c.setConnectTimeout(30000); c.setReadTimeout(60000);
        c.setRequestProperty("Authorization", "Bearer " + apiKey);
        c.setRequestProperty("Content-Type", "multipart/form-data; boundary=" + boundary);
        try (OutputStream out = c.getOutputStream(); InputStream in = context.getContentResolver().openInputStream(uri)) {
            write(out, "--"+boundary+"\r\nContent-Disposition: form-data; name=\"purpose\"\r\n\r\nuser_data\r\n");
            write(out, "--"+boundary+"\r\nContent-Disposition: form-data; name=\"file\"; filename=\""+safe(fileName)+"\"\r\nContent-Type: "+mime+"\r\n\r\n");
            if (in == null) throw new IOException("A fájl nem olvasható");
            byte[] buf = new byte[8192]; int n; while((n=in.read(buf))!=-1) out.write(buf,0,n);
            write(out, "\r\n--"+boundary+"--\r\n");
        }
        String body = readResponse(c);
        if (c.getResponseCode()/100 != 2) throw new IOException("OpenAI fájlfeltöltési hiba: " + compactError(body));
        return new JSONObject(body).getString("id");
    }
