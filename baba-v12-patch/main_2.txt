    private void analyzeContractIntoJob(Uri uri) {
        JobDialogState state = activeJobDialog;
        if (state == null) return;
        String key=SecureStore.getApiKey(this); if(key.isEmpty()){showSettings();return;}
        state.aiStatus.setText("AI elemzés folyamatban…"); state.aiStatus.setTextColor(roseDark);
        ProgressDialog p=new ProgressDialog(this); p.setTitle("Munkaszerződés elemzése"); p.setMessage("Az AI kiolvassa a munkaviszony adatait…"); p.setIndeterminate(true); p.setCancelable(false); p.show();
        new Thread(() -> {
            try {
                OpenAIClient.EmploymentExtraction e=OpenAIClient.extractEmploymentData(this,uri,key,SecureStore.getModel(this));
                runOnUiThread(() -> {
                    p.dismiss();
                    if (activeJobDialog != state) return;
                    if (!e.employer.isEmpty()) state.employer.setText(e.employer);
                    if (!e.start.isEmpty()) state.start.setText(e.start);
                    if (!e.end.isEmpty()) state.end.setText(e.end); else if (e.ongoing) state.end.setText("");
                    if (e.monthlyGross > 0) state.gross.setText(String.valueOf(e.monthlyGross));
                    if (e.sickPayDays >= 0) state.sick.setText(String.valueOf(e.sickPayDays));
                    if (e.unpaidDays >= 0) state.unpaid.setText(String.valueOf(e.unpaidDays));
                    state.insured.setChecked(e.insured);
                    state.aiStatus.setText("✓ AI kitöltés elkészült" + (e.notes.isEmpty()?"":" • "+e.notes));
                    state.aiStatus.setTextColor(Color.rgb(65,126,94));
                    showHelp("AI kitöltés elkészült", "A felismert adatokat beírtam a mezőkbe. Mentés előtt ellenőrizd őket, és szükség esetén módosítsd kézzel.\n\n" + (e.notes.isEmpty()?"Az AI nem jelzett külön bizonytalanságot.":e.notes));
                });
            } catch(Exception e) { runOnUiThread(() -> {p.dismiss(); if(activeJobDialog==state){state.aiStatus.setText("AI elemzés sikertelen");state.aiStatus.setTextColor(Color.rgb(173,91,67));} showHelp("Elemzési hiba", e.getMessage()==null?"Ismeretlen hiba":e.getMessage());}); }
        }).start();
    }
