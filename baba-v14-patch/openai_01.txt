    private static String createResponseForFile(String fileId, String apiKey, String model, String prompt) throws Exception {
        JSONArray content = new JSONArray();
        content.put(new JSONObject().put("type","input_text").put("text",prompt));

    private static String createResponseForImage(Context context, Uri uri, String mime, String apiKey, String model, String prompt) throws Exception {
        byte[] bytes = readAll(context, uri, 18 * 1024 * 1024);
        String dataUrl = "data:" + mime + ";base64," + Base64.encodeToString(bytes, Base64.NO_WRAP);
        JSONArray content = new JSONArray();
        content.put(new JSONObject().put("type","input_text").put("text",prompt));
        content.put(new JSONObject().put("type","input_image").put("image_url",dataUrl).put("detail","high"));
        return callResponses(content, apiKey, model);
    }
