    private void showSettings() {
        LinearLayout x = new LinearLayout(this); x.setOrientation(LinearLayout.VERTICAL); x.setPadding(dp(18), dp(6), dp(18), 0);
        TextView apiLabel = label("OpenAI API-kulcs"); x.addView(apiLabel);
        EditText api = input("sk-..."); api.setInputType(InputType.TYPE_CLASS_TEXT | InputType.TYPE_TEXT_VARIATION_PASSWORD);
        String current = SecureStore.getApiKey(this); if (!current.isEmpty()) api.setText(current); x.addView(api);
        TextView modelLabel = label("OpenAI modell"); modelLabel.setPadding(0,dp(12),0,0); x.addView(modelLabel);
        EditText model = input("gpt-5.4-mini"); model.setText(SecureStore.getModel(this)); x.addView(model);
        TextView info = text("A kulcs AES-GCM titkosítással, Android Keystore által védett kulccsal kerül mentésre ezen az eszközön. Közvetlen mobil API-használat személyes/teszt célra alkalmas; nyilvánosan terjesztett alkalmazásnál biztonságosabb saját backend használata.", 12, false);
        info.setTextColor(muted); info.setPadding(0,dp(12),0,0); x.addView(info);
        AlertDialog dlg = new AlertDialog.Builder(this).setTitle("Beállítások").setView(x)
                .setPositiveButton("Mentés", null).setNegativeButton("Mégse", null)
                .setNeutralButton("Kulcs törlése", null).create();
        dlg.setOnShowListener(v -> {
            dlg.getButton(AlertDialog.BUTTON_POSITIVE).setOnClickListener(w -> {
                try {
                    SecureStore.saveApiKey(this, api.getText().toString());
                    SecureStore.saveModel(this, model.getText().toString().trim().isEmpty()?"gpt-5.4-mini":model.getText().toString().trim());
                    toast("Beállítások elmentve"); dlg.dismiss(); rebuildUi();
                } catch (Exception e) { toast("A kulcs mentése nem sikerült: " + e.getMessage()); }
            });
            dlg.getButton(AlertDialog.BUTTON_NEUTRAL).setOnClickListener(w -> {
                try { SecureStore.saveApiKey(this, ""); toast("API-kulcs törölve"); dlg.dismiss(); rebuildUi(); }
                catch (Exception e) { toast("Törlési hiba"); }
            });
        });
        dlg.show();
    }
