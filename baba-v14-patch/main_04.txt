    private void updateDocumentLog(JobDialogState state){if(state.evidence.isEmpty()){state.documentLog.setText("Még nincs feldolgozott dokumentum.");return;}StringBuilder b=new StringBuilder("Feldolgozott dokumentumok: "+state.evidence.size());for(String e:state.evidence)b.append("\n• ").append(e);state.documentLog.setText(b.toString());}
    private void selectSpinner(Spinner spinner,String value){if(value==null)return;for(int i=0;i<spinner.getCount();i++)if(value.equalsIgnoreCase(String.valueOf(spinner.getItemAtPosition(i)))){spinner.setSelection(i);return;}}
