    public static BatchExtraction extractEmploymentBatch(Context context, List<Uri> uris, String userNotes, String apiKey, String model) throws Exception {
        JSONArray content = new JSONArray();
        content.put(new JSONObject().put("type","input_text").put("text", batchPrompt(userNotes, uris == null ? 0 : uris.size())));
        if (uris != null) for (Uri uri : uris) {
            String mime = context.getContentResolver().getType(uri); if (mime == null) mime = "application/octet-stream";
            if (mime.startsWith("image/")) {
                byte[] bytes = readAll(context, uri, 18 * 1024 * 1024);
                String dataUrl = "data:" + mime + ";base64," + Base64.encodeToString(bytes, Base64.NO_WRAP);
                content.put(new JSONObject().put("type","input_image").put("image_url",dataUrl).put("detail","high"));
            } else {
                String fileName = getFileName(context.getContentResolver(), uri);
                String fileId = uploadFile(context, uri, fileName, mime, apiKey);
                content.put(new JSONObject().put("type","input_file").put("file_id",fileId));
            }
        }
        return parseBatchExtraction(callResponses(content, apiKey, model));
    }

    public static EmploymentExtraction extractEmploymentData(Context context, Uri uri, String apiKey, String model) throws Exception {
        String mime = context.getContentResolver().getType(uri);
        if (mime == null) mime = "application/octet-stream";
        String prompt = structuredPrompt();
        String text;
        if (mime.startsWith("image/")) {
            text = createResponseForImage(context, uri, mime, apiKey, model, prompt);
        } else {
            String fileName = getFileName(context.getContentResolver(), uri);
            String fileId = uploadFile(context, uri, fileName, mime, apiKey);
            text = createResponseForFile(fileId, apiKey, model, prompt);
        }
        return parseEmploymentExtraction(text);
    }
